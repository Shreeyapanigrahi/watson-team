# Q11: Knowledge space health

**Question:** For each knowledge space: how many active documents, their average quality, the latest ingest, and how many archived?

```sparql
PREFIX Watson: <http://watson.local/pharma#>
PREFIX rdf:  <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd:  <http://www.w3.org/2001/XMLSchema#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
SELECT ?space (COALESCE(?docs, 0) AS ?documents) (ROUND(?avgQuality * 1000) / 1000 AS ?averageQuality) ?latestIngest (COALESCE(?arch, 0) AS ?archived)
WHERE {
  ?sp a Watson:KnowledgeSpace ; Watson:spaceName ?space .
  OPTIONAL {                                     # a space with no active document still appears, with 0
    SELECT ?sp (COUNT(DISTINCT ?d) AS ?docs) (AVG(?q) AS ?avgQuality) (MAX(?ing) AS ?latestIngest)
    WHERE { ?sp Watson:containsDocument ?d .
            ?d Watson:lifecycleState "ACTIVE" ; Watson:quality_score ?q ; Watson:ingest_timestamp ?ing . }
    GROUP BY ?sp
  }
  OPTIONAL {
    SELECT ?sp (COUNT(DISTINCT ?a) AS ?arch)
    WHERE { ?sp Watson:containsDocument ?a . ?a Watson:lifecycleState "ARCHIVED" }
    GROUP BY ?sp
  }
}
ORDER BY ?space
```
